Beispiel für die InvoiceView.js:

if (saveBtn) {

    };
}
if (printBtn) {

        e.stopPropagation();
        if (typeof window.executePrint === 'function') window.executePrint('print');
    };
}
if (closeBtn) {
    closeBtn.onclick = (e) => {
        e.preventDefault();
        e.stopPropagation();
        this.closePdfPreview();
    };
}
