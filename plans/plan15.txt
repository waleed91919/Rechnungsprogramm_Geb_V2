Da wir nun das extrem sichere .onclick in der InvoiceView verwenden (welches Duplikate unmöglich macht), ist es jetzt absolut sicher und sogar NOTWENDIG, das UI im finally-Block hart zu resetten!

            btn.classList.remove('opacity-50', 'pointer-events-none');
        }
    });

    if (window.invoiceView && typeof window.invoiceView.bindPdfModalControls === 'function') {
        window.invoiceView.bindPdfModalControls();

    // 3. Fokus zurückholen

}
